# skrypt: create_persistent_profile_with_timeout.py
import subprocess
import os
import time
import threading

# --- Konfiguracja ---
PROFILE_PATH = r"C:\automatyzacja\chrome-profil"
CHROME_PATH = r"C:\Program Files\Google\Chrome\Application\chrome.exe"
LOGIN_URL = "https://myaccount.microsoft.com/"
LOGIN_TIMEOUT_SECONDS = 60 # Czas na ręczne zalogowanie

# --- Funkcja zamykająca proces w tle ---
def close_process_after_timeout(process, timeout):
    """
    Funkcja, która będzie działać w osobnym wątku.
    Czeka na określony czas, a następnie próbuje zamknąć proces,
    jeśli ten wciąż jest aktywny.
    """
    # Czekaj, aż główny wątek zakończy proces lub upłynie czas
    process.wait(timeout=timeout)
    
    # Jeśli po upływie czasu proces nadal działa, zamknij go
    if process.poll() is None:
        print(f"\n[Timeout] Upłynęło {timeout} sekund. Automatycznie zamykam przeglądarkę.")
        try:
            process.terminate() # Bezpieczniejsza metoda
            process.wait(5) # Daj mu chwilę na zamknięcie
            if process.poll() is None:
                process.kill() # Metoda ostateczna
        except Exception as e:
            print(f"Wystąpił błąd podczas próby zamknięcia procesu: {e}")

# --- Główna logika skryptu ---
if not os.path.exists(CHROME_PATH):
    print(f"BŁĄD: Nie znaleziono przeglądarki Chrome w ścieżce: {CHROME_PATH}")
    exit()

if not os.path.exists(PROFILE_PATH):
    os.makedirs(PROFILE_PATH)
    print(f"Utworzono folder dla profilu: {PROFILE_PATH}")

command = [
    CHROME_PATH,
    f"--user-data-dir={PROFILE_PATH}",
    LOGIN_URL
]

print("\nOtwieram przeglądarkę z dedykowanym profilem.")
print("PROSZĘ ZALOGOWAĆ SIĘ NA KONTO MICROSOFT w ciągu 60 sekund.")
print("Upewnij się, że zaznaczyłeś/aś opcję 'Nie wylogowuj mnie' (Stay signed in).")
print("\nPo pomyślnym zalogowaniu możesz od razu zamknąć okno przeglądarki.")

# Uruchom proces przeglądarki
process = subprocess.Popen(command)

# Uruchom wątek, który będzie monitorował czas
timeout_thread = threading.Thread(
    target=close_process_after_timeout,
    args=(process, LOGIN_TIMEOUT_SECONDS),
    daemon=True  # Ustaw jako daemon, aby wątek zakończył się wraz z głównym skryptem
)

try:
    timeout_thread.start()
    
    # Główny wątek czeka tutaj, aż proces przeglądarki się zakończy
    # (albo przez zamknięcie przez użytkownika, albo przez wątek timeoutu)
    process.wait()

    # Sprawdź, czy proces nadal działa (co by oznaczało, że to wątek go zamknął)
    if timeout_thread.is_alive():
        print("\nPrzeglądarka została zamknięta przez użytkownika. Kontynuuję.")
        # Tutaj moglibyśmy formalnie "dołączyć" do wątku, ale jako daemon nie jest to konieczne
    
    # Dajmy chwilę na pełne zapisanie danych profilu
    time.sleep(2)
    print("Profil został utworzony i zawiera zapisaną sesję logowania.")

except Exception as e:
    print(f"Wystąpił nieoczekiwany błąd: {e}")
finally:
    # Upewnij się, że proces na pewno jest zamknięty na końcu
    if process.poll() is None:
        process.kill()

# skrypt: get_token_with_persistent_profile.py
import os
import shutil
from urllib.parse import urlparse, parse_qs

import requests
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

# --- Konfiguracja ---
CLIENT_ID = "twoj_client_id"
TENANT_ID = "twoj_tenant_id"
REDIRECT_URI = "twoj_redirect_uri"
SCOPE = "user.read offline_access email"

# Użyj tej samej stałej ścieżki, co w kroku 1
PERSISTENT_PROFILE_PATH = r"C:\automatyzacja\chrome-profil"

AUTH_URI = (
    f"https://login.microsoftonline.com/{TENANT_ID}/oauth2/v2.0/authorize?"
    f"client_id={CLIENT_ID}&response_type=code&redirect_uri={REDIRECT_URI}&"
    f"response_mode=query&scope={SCOPE}"
)

def get_token_from_persistent_session():
    """Pobiera token, ponownie wykorzystując istniejący profil Chrome."""
    
    if not os.path.exists(PERSISTENT_PROFILE_PATH):
        print("BŁĄD: Folder profilu nie istnieje!")
        print(f"Proszę najpierw uruchomić skrypt 'create_persistent_profile.py', aby go stworzyć i się zalogować.")
        return None

    options = webdriver.ChromeOptions()
    options.add_argument(f"--user-data-dir={PERSISTENT_PROFILE_PATH}")
    options.add_argument("--headless=new")
    options.add_argument("--window-size=1920,1080")
    options.add_argument("--no-sandbox")
    options.add_argument("--disable-dev-shm-usage")
    options.add_argument("--disable-gpu")

    service = Service(ChromeDriverManager().install())
    driver = None
    authorization_code = None

    try:
        print("Uruchamiam przeglądarkę w tle z zapisaną sesją...")
        driver = webdriver.Chrome(service=service, options=options)
        driver.get(AUTH_URI)

        wait = WebDriverWait(driver, 15)
        wait.until(EC.url_contains("code="))
        
        final_url = driver.current_url
        parsed_url = urlparse(final_url)
        authorization_code = parse_qs(parsed_url.query).get('code', [None])[0]

        if authorization_code:
            print("Pomyślnie przechwycono kod autoryzacyjny.")
        else:
            print("Błąd krytyczny: Nie udało się uzyskać kodu mimo poprawnego przekierowania.")

    except Exception as e:
        print(f"Wystąpił błąd podczas automatyzacji: {e}")
        print("Możliwa przyczyna: Sesja logowania w profilu wygasła.")
        print("Uruchom ponownie skrypt 'create_persistent_profile.py', aby odświeżyć logowanie.")
    finally:
        if driver:
            driver.quit()

    if not authorization_code:
        return None

    # Wymiana kodu na token
    token_endpoint = f"https://login.microsoftonline.com/{TENANT_ID}/oauth2/v2.0/token"
    payload = {
        'client_id': CLIENT_ID, 'scope': SCOPE, 'code': authorization_code,
        'redirect_uri': REDIRECT_URI, 'grant_type': 'authorization_code',
    }
    
    print("Wymieniam kod na token dostępu...")
    response = requests.post(token_endpoint, data=payload)
    token_data = response.json()

    if "access_token" in token_data:
        print("Sukces! Uzyskano token dostępu.")
        return token_data["access_token"]
    else:
        print("\nNie udało się uzyskać tokenu. Odpowiedź z serwera:", token_data)
        return None

# --- Uruchomienie skryptu ---
if __name__ == "__main__":
    access_token = get_token_from_persistent_session()
    if access_token:
        print("\nMożesz teraz użyć uzyskanego tokenu do rezerwacji biurka.")
